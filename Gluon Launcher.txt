class AppRepository(private val context: Context) {

    private val appContext = context.applicationContext
    private val packageManager = appContext.packageManager
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Default + CoroutineName("AppRepo"))
    private val dataStore = appContext.dataStore

    private val _appsFlow = MutableStateFlow<List<AppModel>>(emptyList())
    val appsFlow: StateFlow<List<AppModel>> = _appsFlow.asStateFlow()

    private var packageReceiver: BroadcastReceiver? = null

    init {
        scope.launch { refreshApps() }
        observePackageChanges()
    }

    private suspend fun loadAppIcon(info: android.content.pm.ResolveInfo): ImageBitmap? {
        return withContext(Dispatchers.IO) {
            try {
                val drawable = info.loadIcon(packageManager)
                val w = drawable.intrinsicWidth.coerceAtLeast(1)
                val h = drawable.intrinsicHeight.coerceAtLeast(1)
                val bmp = Bitmap.createBitmap(w, h, Bitmap.Config.ARGB_8888)
                val canvas = Canvas(bmp)
                drawable.setBounds(0, 0, canvas.width, canvas.height)
                drawable.draw(canvas)
                bmp.asImageBitmap()
            } catch (_: Exception) {
                null
            }
        }
    }

    suspend fun refreshApps() {
        withContext(Dispatchers.IO) {
            val hidden = dataStore.getHiddenApps()
            val usage = dataStore.getUsageCounts()
            val intent = Intent(Intent.ACTION_MAIN, null).apply {
                addCategory(Intent.CATEGORY_LAUNCHER)
            }
            val resolveInfos = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
                packageManager.queryIntentActivities(intent, PackageManager.ResolveInfoFlags.of(0L))
            } else {
                @Suppress("DEPRECATION")
                packageManager.queryIntentActivities(intent, 0)
            }
            val mapped = resolveInfos.mapNotNull { info ->
                val pkgName = info.activityInfo.packageName
                if (pkgName == appContext.packageName) return@mapNotNull null
                if (pkgName in hidden) return@mapNotNull null
                val installTime = try {
                    packageManager.getPackageInfo(pkgName, 0).firstInstallTime
                } catch (_: Exception) { 0L }

                val iconBitmap = loadAppIcon(info)
                val appLabel = info.loadLabel(packageManager).toString()

                AppModel(
                    label = appLabel,
                    packageName = pkgName,
                    installTime = installTime,
                    isSystem = (info.activityInfo.applicationInfo.flags and
                            android.content.pm.ApplicationInfo.FLAG_SYSTEM) != 0,
                    usageCount = usage[pkgName] ?: 0,
                    iconBitmap = iconBitmap
                )
            }
            val uniqueMapped = mapped.distinctBy { it.packageName }
            val sortedList = uniqueMapped.sortedBy { it.label.lowercase() }
            _appsFlow.update { sortedList }
        }
    }

    suspend fun hideApp(packageName: String) {
        withContext(Dispatchers.IO) {
            dataStore.addHiddenApp(packageName)
        }
        refreshApps()
    }

    suspend fun unhideApp(packageName: String) {
        withContext(Dispatchers.IO) {
            dataStore.removeHiddenApp(packageName)
        }
        refreshApps()
    }

    suspend fun getHiddenApps(): Set<String> {
        return withContext(Dispatchers.IO) {
            dataStore.getHiddenApps()
        }
    }

    suspend fun getHiddenAppModels(): Map<String, AppModel> = withContext(Dispatchers.IO) {
        val pkgNames = dataStore.getHiddenApps()
        val result = mutableMapOf<String, AppModel>()
        for (pkg in pkgNames) {
            try {
                val appInfo = packageManager.getApplicationInfo(pkg, 0)
                val label = packageManager.getApplicationLabel(appInfo).toString()
                val drawable = packageManager.getApplicationIcon(appInfo)
                val bitmap = try {
                    val w = drawable.intrinsicWidth.coerceAtLeast(1)
                    val h = drawable.intrinsicHeight.coerceAtLeast(1)
                    val bmp = Bitmap.createBitmap(w, h, Bitmap.Config.ARGB_8888)
                    val canvas = Canvas(bmp)
                    drawable.setBounds(0, 0, canvas.width, canvas.height)
                    drawable.draw(canvas)
                    bmp.asImageBitmap()
                } catch (_: Exception) { null }
                result[pkg] = AppModel(
                    label = label,
                    packageName = pkg,
                    iconBitmap = bitmap
                )
            } catch (_: Exception) { }
        }
        result
    }

    private fun observePackageChanges() {
        if (packageReceiver != null) return
        val filter = IntentFilter().apply {
            addAction(Intent.ACTION_PACKAGE_ADDED)
            addAction(Intent.ACTION_PACKAGE_REMOVED)
            addAction(Intent.ACTION_PACKAGE_CHANGED)
            addDataScheme("package")
        }
        packageReceiver = object : BroadcastReceiver() {
            override fun onReceive(context: Context?, intent: Intent?) {
                val pkgName = intent?.data?.schemeSpecificPart ?: return
                if (pkgName == appContext.packageName) return
                scope.launch { refreshApps() }
            }
        }
        val flags = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            ContextCompat.RECEIVER_NOT_EXPORTED
        } else if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            ContextCompat.RECEIVER_EXPORTED
        } else {
            0
        }
        ContextCompat.registerReceiver(appContext, packageReceiver, filter, flags)
    }

    fun onDestroy() {
        try {
            packageReceiver?.let { appContext.unregisterReceiver(it) }
        } catch (_: Exception) {}
        packageReceiver = null
        scope.cancel()
    }

    suspend fun incrementUsage(packageName: String) {
        withContext(Dispatchers.IO) {
            dataStore.incrementUsageCount(packageName)
            _appsFlow.update { currentApps ->
                currentApps.map {
                    if (it.packageName == packageName) it.copy(usageCount = it.usageCount + 1) else it
                }
            }
        }
    }
}

package com.gluon.launcher.core.utils

import android.content.Context
import android.net.Uri
import java.io.File
import java.io.FileOutputStream

object AvatarUtils {
    fun processImageUri(context: Context, uri: Uri): File? {
        return try {
            val appContext = context.applicationContext
            val file = File(appContext.cacheDir, "gluon_upload_${System.currentTimeMillis()}.jpg")
            val inputStream = appContext.contentResolver.openInputStream(uri) ?: return null
            inputStream.use { input ->
                FileOutputStream(file).use { output ->
                    input.copyTo(output)
                }
            }
            if (file.exists() && file.length() > 0) file else {
                file.delete()
                null
            }
        } catch (e: Exception) {
            e.printStackTrace()
            null
        }
    }
}

class MainViewModelFactory(private val application: Application) : ViewModelProvider.Factory {
    override fun <T : ViewModel> create(modelClass: Class<T>): T {
        if (modelClass.isAssignableFrom(MainViewModel::class.java)) {
            val authManager = AuthManager(application)
            val themeManager = ThemeManager(application)
            val profileManager = ProfileManager(authManager)
            val appRepository = AppRepository(application)
            return MainViewModel(application, authManager, themeManager, profileManager, appRepository) as T
        }
        throw IllegalArgumentException("Unknown ViewModel class")
    }

import androidx.compose.material3.Typography
import androidx.compose.ui.text.TextStyle
import androidx.compose.ui.text.font.FontFamily
import androidx.compose.ui.text.font.FontWeight

import java.util.concurrent.ConcurrentHashMap

private val typographyCache = ConcurrentHashMap<Float, Typography>()

fun getGluonTypography(scale: Float = 1.0f): Typography {

// app/src/main/java/com/gluon/launcher/core/auth/ProfileManager.kt
package com.gluon.launcher.core.auth

import android.content.Context
import android.net.Uri
import coil.annotation.ExperimentalCoilApi
import coil.imageLoader
import com.gluon.launcher.core.data.RetrofitClient
import com.gluon.launcher.core.data.UserItem
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.MediaType.Companion.toMediaTypeOrNull
import okhttp3.MultipartBody
import okhttp3.RequestBody
import okhttp3.RequestBody.Companion.asRequestBody
import okhttp3.RequestBody.Companion.toRequestBody
import java.io.File
import java.io.FileOutputStream
import java.io.IOException
import java.net.SocketTimeoutException
import java.net.UnknownHostException

class ProfileManager(private val authManager: AuthManager) {

    val currentUser = authManager.currentUser

    private fun getUserId(): String? = authManager.currentUser.value?.id

    private fun handleNetworkError(e: Exception): String {
        return when (e) {
            is SocketTimeoutException -> "Превышено время ожидания. Проверьте подключение."
            is UnknownHostException -> "Нет сети. Проверьте интернет-соединение."
            is IOException -> "Ошибка сети: ${e.message}"
            else -> e.message ?: "Неизвестная ошибка"
        }
    }

    @OptIn(ExperimentalCoilApi::class)
    suspend fun deleteAvatar(context: Context): Result<UserItem> = withContext(Dispatchers.IO) {
        val userId = getUserId() ?: return@withContext Result.failure(Exception("Ошибка: отсутствует ID пользователя"))
        val appContext = context.applicationContext

        try {
            authManager.currentUser.value?.getAvatarUrl()?.let { url ->
                appContext.imageLoader.diskCache?.remove(url)
                appContext.imageLoader.memoryCache?.remove(coil.memory.MemoryCache.Key(url))
            }

            val response = RetrofitClient.api.forceDeleteAvatar(userId, mapOf("avatar" to ""))

            if (response.isSuccessful && response.body() != null) {
                val updatedUser = response.body()!!.copy(avatar = "")
                authManager.updateLocalUser(updatedUser)
                Result.success(updatedUser)
            } else {
                Result.failure(Exception("Ошибка сервера при удалении аватара"))
            }
        } catch (e: Exception) {
            Result.failure(Exception(handleNetworkError(e)))
        }
    }

    @OptIn(ExperimentalCoilApi::class)
    suspend fun updateProfile(
        context: Context,
        newName: String,
        newGluonId: String,
        gender: String,
        bio: String,
        imageUri: Uri? = null
    ): Result<UserItem> = withContext(Dispatchers.IO) {
        val userId = getUserId() ?: return@withContext Result.failure(Exception("Ошибка: отсутствует ID пользователя"))
        val appContext = context.applicationContext
        var tempFile: File? = null

        try {
            val mediaType = "text/plain".toMediaTypeOrNull()
            val partMap = mutableMapOf<String, RequestBody>()

            partMap["full_name"] = newName.trim().toRequestBody(mediaType)
            partMap["gluon_id"] = newGluonId.trim().toRequestBody(mediaType)
            partMap["gender"] = gender.toRequestBody(mediaType)
            partMap["bio"] = bio.trim().toRequestBody(mediaType)

            var avatarPart: MultipartBody.Part? = null

            imageUri?.let { uri ->
                val file = File(appContext.cacheDir, "gluon_update_temp_${System.currentTimeMillis()}.jpg")
                tempFile = file
                appContext.contentResolver.openInputStream(uri)?.use { input ->
                    FileOutputStream(file).use { output -> input.copyTo(output) }
                }

                if (file.exists() && file.length() > 0) {
                    val reqFile = file.asRequestBody("image/jpeg".toMediaTypeOrNull())
                    avatarPart = MultipartBody.Part.createFormData("avatar", "avatar.jpg", reqFile)

                    authManager.currentUser.value?.getAvatarUrl()?.let { oldUrl ->
                        appContext.imageLoader.diskCache?.remove(oldUrl)
                        appContext.imageLoader.memoryCache?.remove(coil.memory.MemoryCache.Key(oldUrl))
                    }
                }
            }

            val response = RetrofitClient.api.updateProfile(userId, partMap, avatarPart)

            if (response.isSuccessful && response.body() != null) {
                val updatedUser = response.body()!!
                authManager.updateLocalUser(updatedUser)
                Result.success(updatedUser)
            } else {
                Result.failure(Exception("Ошибка при обновлении профиля"))
            }
        } catch (e: Exception) {
            Result.failure(Exception(handleNetworkError(e)))
        } finally {
            tempFile?.delete()
        }
    }

    suspend fun changePassword(
        oldPass: String,
        newPass: String,
        confirmPass: String
    ): Result<UserItem> = withContext(Dispatchers.IO) {
        val userId = getUserId() ?: return@withContext Result.failure(Exception("Ошибка: отсутствует ID пользователя"))
        try {
            val body = mapOf(
                "oldPassword" to oldPass,
                "password" to newPass,
                "passwordConfirm" to confirmPass
            )
            val response = RetrofitClient.api.updatePasswordJson(userId, body)

            if (response.isSuccessful && response.body() != null) {
                val updated = response.body()!!
                authManager.updateLocalUser(updated)
                Result.success(updated)
            } else {
                val errorMsg = when (response.code()) {
                    400 -> "Неверный текущий пароль"
                    else -> "Ошибка сервера при смене пароля"
                }
                Result.failure(Exception(errorMsg))
            }
        } catch (e: Exception) {
            Result.failure(Exception(handleNetworkError(e)))
        }
    }

    suspend fun deleteAccount(): Result<Unit> = withContext(Dispatchers.IO) {
        val userId = getUserId() ?: return@withContext Result.failure(Exception("Ошибка: отсутствует ID пользователя"))
        try {
            val res = RetrofitClient.api.deleteAccount(userId)
            if (res.isSuccessful) {
                authManager.clearData()
                Result.success(Unit)
            } else {
                Result.failure(Exception("Не удалось удалить аккаунт"))
            }
        } catch (e: Exception) {
            Result.failure(Exception(handleNetworkError(e)))
        }
    }
}

                    .fillMaxSize()

                                    text = fullName.take(1).uppercase(),

                                    color = MaterialTheme.colorScheme.primary,
                                    fontWeight = FontWeight.Black
                                )
                            }
                        }
                    }

                    val hasSelectedNew = selectedImageUri != null
                    val canDelete = hasPhotoOnServer || hasSelectedNew

                        onClick = {
                            if (!isLoading) {
                                when {
                                    hasSelectedNew -> selectedImageUri = null

    var newPass by remember { mutableStateOf("") }
    var confirmPass by remember { mutableStateOf("") }
    var visible by remember { mutableStateOf(false) }

    val isConfirmError = confirmPass.isNotEmpty() && newPass != confirmPass
    val isSameAsOld = newPass.isNotEmpty() && oldPass == newPass && oldPass.isNotEmpty()

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .imePadding()
            .navigationBarsPadding()
            .padding((24 * scale).dp)
            .verticalScroll(rememberScrollState())
    ) {

            verticalAlignment = Alignment.CenterVertically
        ) {

                fontWeight = FontWeight.Black
            )
            TextButton(onClick = onDismiss) { Text("Отмена") }
        }
        Text(

            style = MaterialTheme.typography.bodyMedium,

            modifier = Modifier.padding(top = (4 * scale).dp)
        )

            label = { Text("Текущий пароль") },

            shape = MaterialTheme.shapes.medium,

                unfocusedContainerColor = MaterialTheme.colorScheme.surfaceVariant,
                focusedIndicatorColor = MaterialTheme.colorScheme.primary,

        dampingRatio = Spring.DampingRatioNoBouncy,
        stiffness = Spring.StiffnessMedium
    )

    // Для скейлинга (кнопки, нажатия)
    val scale = spring<Float>(
        dampingRatio = Spring.DampingRatioHighBouncy,
        stiffness = Spring.StiffnessLow
    )

package com.gluon.launcher.core.data

import android.os.Parcelable

import kotlinx.parcelize.Parcelize
import kotlinx.serialization.SerialName
import kotlinx.serialization.Serializable

private const val GLUON_DOMAIN = "https://gluoncore.ddns.net"

@Keep
@Serializable
@Parcelize
data class UserItem(
    @SerialName("id") val id: String = "",
    @SerialName("email") val email: String = "",
    @SerialName("verified") val verified: Boolean = false,
    @SerialName("full_name") val fullName: String = "",
    @SerialName("gender") val gender: String = "Male",
    @SerialName("gluon_id") val gluonId: String = "",
    @SerialName("bio") val bio: String = "",
    @SerialName("avatar") val avatar: String = "",
    @SerialName("tg_auth_id") val tgAuthId: String = "",
    @SerialName("collectionId") val collectionId: String = "",
    @SerialName("collectionName") val collectionName: String = "Gluon_Database",
    @SerialName("created") val created: String = "",
    @SerialName("updated") val updated: String = ""
) : Parcelable {

    fun getAvatarUrl(): String? {
        if (avatar.isBlank() || collectionName.isBlank() || id.isBlank()) return null
        val numericTimestamp = updated.filter { it.isDigit() }
        val version = if (numericTimestamp.length >= 13) numericTimestamp.takeLast(13) else numericTimestamp
        return "$GLUON_DOMAIN/api/files/$collectionName/$id/$avatar?v=$version"
    }

    fun isProfileComplete(): Boolean =
        fullName.trim().isNotBlank() && gluonId.trim().isNotBlank()
}

@Keep

@Parcelize

@Composable
fun NormalHomeGrid(
    appsList: List<String?>,
    allApps: List<AppModel>,
    showLabels: Boolean,
    showIconBorder: Boolean,
    totalRows: Int,
    columns: Int,
    iconScale: Float = 1f,
    onAppClick: (AppModel) -> Unit,
    onRemoveApp: (Int) -> Unit,
    textColor: Color      // ✅ новый параметр для цвета текста
) {
    val scale = LocalScale.current
    val horizontalPadding = (16 * scale).dp
    val horizontalSpacing = (when (columns) { 3 -> 16; 4 -> 12; 5 -> 8; else -> 12 } * scale).dp
    val verticalSpacing = (18 * scale).dp

    Column(
        modifier = Modifier.fillMaxSize(),
        verticalArrangement = Arrangement.spacedBy(verticalSpacing)
    ) {
        for (row in 0 until totalRows) {
            Row(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
                    .padding(horizontal = horizontalPadding),
                horizontalArrangement = Arrangement.spacedBy(horizontalSpacing)
            ) {
                for (col in 0 until columns) {
                    val index = row * columns + col
                    val pkg = if (index < appsList.size) appsList[index] else null
                    val app = if (pkg != null) allApps.find { it.packageName == pkg } else null
                    Box(
                        modifier = Modifier
                            .weight(1f)
                            .fillMaxSize()
                    ) {
                        if (app != null) {
                            AppIconItem(
                                app = app,
                                scale = iconScale,
                                showLabel = showLabels,
                                showIconBorder = showIconBorder,
                                isInGrid = true,
                                onAppClick = { onAppClick(app) },
                                modifier = Modifier.fillMaxSize(),
                                contextMenuOffset = androidx.compose.ui.unit.DpOffset((16 * scale).dp, 0.dp),
                                textColor = textColor,    // ✅ передаём контрастный цвет
                                onContextMenu = { dismiss ->
                                    Column(
                                        modifier = Modifier.width((180 * scale).dp),
                                        verticalArrangement = Arrangement.spacedBy(4.dp)
                                    ) {
                                        GluonPopupMenuCard(
                                            onClick = { dismiss(); onRemoveApp(index) },
                                            color = MaterialTheme.colorScheme.errorContainer
                                        ) {
                                            Icon(Icons.Default.Delete, null, tint = MaterialTheme.colorScheme.onErrorContainer)
                                            Spacer(Modifier.width(8.dp))
                                            GluonMarqueeText("Убрать с рабочего стола", color = MaterialTheme.colorScheme.onErrorContainer)
                                        }
                                    }
                                }
                            )
                        } else {
                            Box(
                                modifier = Modifier
                                    .fillMaxSize()
                                    .padding((8 * scale).dp)
                                    .clip(CircleShape)
                            )
                        }
                    }
                }
            }
        }
    }

package com.gluon.launcher.core.auth

import android.content.Context
import android.content.SharedPreferences

import android.util.Log

class AuthManager(context: Context) {

    private val appContext = context.applicationContext
    private val sessionMutex = Mutex()

    private val exceptionHandler = CoroutineExceptionHandler { _, throwable ->
        Log.e("AuthManager", "Unhandled coroutine exception", throwable)
    }

    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO + exceptionHandler)

    private val masterKeyAlias: String by lazy {
        MasterKeys.getOrCreate(MasterKeys.AES256_GCM_SPEC)
    }

    private val prefs: SharedPreferences by lazy {
        EncryptedSharedPreferences.create(
            "gluon_secure_prefs",
            masterKeyAlias,
            appContext,
            EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
            EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM
        )
    }

    private val _currentUser = MutableStateFlow<UserItem?>(null)
    val currentUser: StateFlow<UserItem?> = _currentUser.asStateFlow()

    private val _isGuest = MutableStateFlow(false)
    val isGuest: StateFlow<Boolean> = _isGuest.asStateFlow()

    private val _isReady = MutableStateFlow(false)
    val isReady: StateFlow<Boolean> = _isReady.asStateFlow()

    init {
        scope.launch {
            loadSession()
            launch { validateSession() }
            _isReady.value = true
        }
    }

    fun isLoggedIn(): Boolean = _currentUser.value != null && !_isGuest.value

    fun isGuestMode(): Boolean = prefs.getBoolean("is_guest", false)

    private fun handleNetworkError(e: Exception): String {
        if (e is HttpException) {
            return when (e.code()) {
                400 -> "Некорректный запрос. Проверьте данные."
                401 -> "Сессия устарела. Пожалуйста, войдите заново."
                404 -> "Данные не найдены на сервере."
                500 -> "Внутренняя ошибка сервера. Попробуйте позже."
                else -> "Ошибка сервера (${e.code()})"
            }
        }
        return when (e) {
            is SocketTimeoutException -> "Превышено время ожидания. Проверьте подключение."
            is UnknownHostException -> "Нет сети. Проверьте интернет-соединение."
            is IOException -> if (e.message?.contains("SSL") == true) "Ошибка безопасности соединения."
            else "Ошибка сети: ${e.message}"
            else -> e.message ?: "Неизвестная ошибка"
        }
    }

    private suspend fun loadSession() = withContext(Dispatchers.IO) {
        try {
            val savedToken = prefs.getString("auth_token", null)
            val savedId = prefs.getString("user_id", null)
            val savedIsGuest = prefs.getBoolean("is_guest", false)

            if (!savedToken.isNullOrEmpty() && !savedId.isNullOrEmpty()) {
                RetrofitClient.setAuthToken(savedToken)
                _currentUser.value = UserItem(
                    id = savedId,
                    fullName = prefs.getString("user_name", "") ?: "",
                    email = prefs.getString("user_email", "") ?: "",
                    gluonId = prefs.getString("gluon_id", "") ?: "",
                    collectionName = "Gluon_Database",
                    avatar = prefs.getString("user_avatar", "") ?: "",
                    verified = prefs.getBoolean("user_verified", false),
                    bio = prefs.getString("user_bio", "") ?: "",
                    gender = prefs.getString("gender", "Male") ?: "Male"
                )
                _isGuest.value = false
            } else if (savedIsGuest) {
                _isGuest.value = true
            }
        } catch (e: Exception) {
            Log.e("AuthManager", "Failed to load session, clearing data", e)
            clearData()
        }
    }

    private suspend fun validateSession() {
        val email = _currentUser.value?.email ?: return
        try {
            val response = RetrofitClient.api.getUserByEmail("(email='${email.trim()}')")
            if (response.code() == 401) {
                Log.w("AuthManager", "Token expired (401), clearing session")
                clearData()
                _isReady.value = true
            }
        } catch (e: Exception) {
            Log.w("AuthManager", "Session validation network error", e)
        }
    }

    suspend fun login(email: String, pass: String): Result<Unit> = withContext(Dispatchers.IO) {
        try {
            val body = mapOf("identity" to email.trim(), "password" to pass.trim())
            val response = RetrofitClient.api.login(body)
            if (response.isSuccessful && response.body() != null) {
                val authData = response.body()!!
                sessionMutex.withLock { saveSession(authData.token, authData.record) }
                Result.success(Unit)
            } else {
                Result.failure(Exception("Неверный email или пароль"))
            }
        } catch (e: Exception) {
            Result.failure(Exception(handleNetworkError(e)))
        }
    }

    suspend fun register(
        email: String,
        pass: String,
        name: String,
        gender: String,
        avatarUri: Uri?
    ): Result<Unit> = withContext(Dispatchers.IO) {
        if (name.trim().isEmpty()) return@withContext Result.failure(Exception("Имя не может быть пустым"))
        val gluonId = email.substringBefore("@") + (1000..9999).random()
        if (gluonId.length !in 5..25 || !gluonId.matches(Regex("^[a-zA-Z0-9_]+$"))) {
            return@withContext Result.failure(Exception("Некорректный gluon_id. Используйте латинские буквы, цифры и _"))
        }

        var tempFile: File? = null
        try {
            val mediaType = "text/plain".toMediaTypeOrNull()
            fun String.toPart() = this.toRequestBody(mediaType)

            var avatarPart: MultipartBody.Part? = null
            avatarUri?.let { uri ->
                tempFile = AvatarUtils.processImageUri(appContext, uri)
                tempFile?.let { file ->
                    val requestFile = file.asRequestBody("image/jpeg".toMediaTypeOrNull())
                    avatarPart = MultipartBody.Part.createFormData("avatar", file.name, requestFile)
                }
            }

            val regResponse = RetrofitClient.api.register(
                email.trim().toPart(),
                pass.trim().toPart(),
                pass.trim().toPart(),
                name.trim().toPart(),
                gender.toPart(),
                gluonId.toPart(),
                "true".toPart(),
                avatarPart
            )

            if (regResponse.isSuccessful) {
                requestVerification(email.trim())
                Result.success(Unit)
            } else {
                val errorMsg = regResponse.errorBody()?.string() ?: "Ошибка регистрации"
                Result.failure(Exception(errorMsg))
            }
        } catch (e: Exception) {
            Result.failure(Exception(handleNetworkError(e)))
        } finally {
            tempFile?.delete()
        }
    }

    suspend fun checkEmailVerificationStatus(email: String): Result<Boolean> = withContext(Dispatchers.IO) {
        try {
            val response = RetrofitClient.api.getUserByEmail("(email='${email.trim()}')")
            if (response.isSuccessful && response.body() != null) {
                val user = response.body()!!.items.firstOrNull()
                val isVerified = user?.verified ?: false
                if (isVerified && user != null) updateLocalUser(user)
                Result.success(isVerified)
            } else Result.failure(Exception("Ошибка сервера"))
        } catch (e: Exception) {
            Result.failure(Exception(handleNetworkError(e)))
        }
    }

    suspend fun sendPasswordResetEmail(email: String): Result<Unit> = withContext(Dispatchers.IO) {
        try {
            val response = RetrofitClient.api.requestPasswordReset(mapOf("email" to email.trim()))
            if (response.isSuccessful) Result.success(Unit)
            else Result.failure(Exception("Не удалось отправить письмо для сброса"))
        } catch (e: Exception) {
            Result.failure(Exception(handleNetworkError(e)))
        }
    }

    suspend fun resendVerificationEmail(email: String): Result<Unit> = withContext(Dispatchers.IO) {
        try {
            val response = RetrofitClient.api.requestVerification(mapOf("email" to email.trim()))
            if (response.isSuccessful) Result.success(Unit)
            else Result.failure(Exception("Не удалось повторно отправить письмо"))
        } catch (e: Exception) {
            Result.failure(Exception(handleNetworkError(e)))
        }
    }

    private suspend fun requestVerification(email: String) {
        try { RetrofitClient.api.requestVerification(mapOf("email" to email.trim())) }
        catch (e: Exception) { Log.e("AuthManager", "Failed to request verification", e) }
    }

    fun saveSession(token: String, user: UserItem) {
        RetrofitClient.setAuthToken(token)
        prefs.edit {
            putString("auth_token", token)
            putString("user_id", user.id)
            putString("user_name", user.fullName)
            putString("user_email", user.email)
            putString("gluon_id", user.gluonId)
            putString("user_avatar", user.avatar)
            putString("user_bio", user.bio)
            putString("gender", user.gender)
            putBoolean("user_verified", user.verified)
            putBoolean("is_guest", false)
        }
        _currentUser.value = user
        _isGuest.value = false
    }

    fun updateLocalUser(user: UserItem) {
        prefs.edit {
            putString("user_name", user.fullName)
            putString("user_avatar", user.avatar)
            putString("gluon_id", user.gluonId)
            putString("user_email", user.email)
            putString("user_bio", user.bio)
            putString("gender", user.gender)
            putBoolean("user_verified", user.verified)
            putBoolean("is_guest", false)
        }
        _currentUser.value = user
        _isGuest.value = false
    }

    fun setGuestMode(isGuest: Boolean) {
        if (isGuest) clearData()
        _isGuest.value = isGuest
        prefs.edit { putBoolean("is_guest", isGuest) }
    }

    fun logout() { clearData() }

    @OptIn(ExperimentalCoilApi::class)
    fun clearData() {
        try {
            appContext.imageLoader.diskCache?.clear()
            appContext.imageLoader.memoryCache?.clear()
        } catch (_: Exception) {}
        RetrofitClient.setAuthToken(null)
        prefs.edit { clear() }
        scope.launch {
            try { appContext.dataStore.edit { it.clear() } }
            catch (e: Exception) { Log.e("AuthManager", "Failed to clear DataStore", e) }
        }
        _currentUser.value = null
        _isGuest.value = false
    }
}

// app/src/main/java/com/gluon/launcher/core/data/RetrofitClient.kt
package com.gluon.launcher.core.data

import kotlinx.serialization.json.Json
import okhttp3.ConnectionPool
import okhttp3.Dns
import okhttp3.Interceptor
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.logging.HttpLoggingInterceptor
import retrofit2.Retrofit
import retrofit2.converter.kotlinx.serialization.asConverterFactory
import java.net.InetAddress
import java.net.UnknownHostException
import java.util.concurrent.TimeUnit
import java.util.concurrent.atomic.AtomicReference

object RetrofitClient {

    const val BASE_URL = "https://gluoncore.ddns.net/"

    // Использование AtomicReference обеспечивает Lock-Free потокобезопасность
    // и исключает накладные расходы на тяжелую синхронизацию в интерцепторе.
    private val authTokenReference = AtomicReference<String?>(null)

    fun setAuthToken(token: String?) {
        authTokenReference.set(token)
    }

    private val jsonConfig = Json {
        ignoreUnknownKeys = true
        coerceInputValues = true
        isLenient = true
        encodeDefaults = true
        prettyPrint = false
    }

    private val customDns = object : Dns {
        override fun lookup(hostname: String): List<InetAddress> {
            return try {
                Dns.SYSTEM.lookup(hostname)
            } catch (_: UnknownHostException) {
                try {
                    InetAddress.getAllByName(hostname).toList()
                } catch (_: Exception) {
                    throw UnknownHostException("GluonCore Network Error: Сервер $hostname не найден в сети.")
                }
            }
        }
    }

    private val authInterceptor = Interceptor { chain ->
        val original = chain.request()
        val requestBuilder = original.newBuilder()
            .header("Accept", "application/json")
            .header("X-Gluon-App", "Android-Core")

        val currentToken = authTokenReference.get()
        currentToken?.let {
            val tokenValue = if (it.startsWith("Bearer ")) it else "Bearer $it"
            requestBuilder.header("Authorization", tokenValue)
        }

        chain.proceed(requestBuilder.build())
    }

    private val okHttpClient: OkHttpClient by lazy {
        val logging = HttpLoggingInterceptor().apply {
            level = HttpLoggingInterceptor.Level.NONE   // Продакшен-режим
        }

        OkHttpClient.Builder()
            .dns(customDns)
            .connectionPool(ConnectionPool(5, 5, TimeUnit.MINUTES))
            .addInterceptor(authInterceptor)
            .addInterceptor(logging)
            .connectTimeout(30, TimeUnit.SECONDS)
            .readTimeout(30, TimeUnit.SECONDS)
            .writeTimeout(30, TimeUnit.SECONDS)
            .retryOnConnectionFailure(true)
            .build()
    }

    private val retrofit: Retrofit by lazy {
        val contentType = "application/json".toMediaType()

        Retrofit.Builder()
            .baseUrl(BASE_URL)
            .client(okHttpClient)
            .addConverterFactory(jsonConfig.asConverterFactory(contentType))
            .build()
    }

    val api: ApiService by lazy {
        retrofit.create(ApiService::class.java)
    }
}

@Composable
private fun PageIndicator(
    pageCount: Int,
    currentPage: Int,
    modifier: Modifier = Modifier
) {
    Row(
        modifier = modifier,
        horizontalArrangement = Arrangement.Center,
        verticalAlignment = Alignment.CenterVertically
    ) {
        repeat(pageCount) { index ->
            val isSelected = index == currentPage
            val size by animateDpAsState(
                targetValue = if (isSelected) 10.dp else 7.dp,
                animationSpec = spring(
                    dampingRatio = Spring.DampingRatioMediumBouncy,
                    stiffness = Spring.StiffnessLow
                )
            )
            Box(
                modifier = Modifier
                    .padding(horizontal = 4.dp)
                    .size(size)
                    .clip(CircleShape)
                    .background(
                        if (isSelected) MaterialTheme.colorScheme.primary
                        else MaterialTheme.colorScheme.onSurface.copy(alpha = 0.4f)
                    )
            )
        }
    }

package com.gluon.launcher.core.utils

/**
 * Нормализует строку: нижний регистр, удаление спецсимволов,
 * свёртка популярных сокращений.
 */
fun String.normalizeForSearch(): String {
    return this.lowercase()
        .replace(Regex("[^a-zа-я0-9 ]"), "")
        .split(" ")
        .joinToString(" ") { word ->
            when (word) {
                "tg" -> "telegram"
                "vk" -> "vkontakte"
                "wa" -> "whatsapp"
                "fb" -> "facebook"
                "ig" -> "instagram"
                "gm" -> "gmail"
                "yt" -> "youtube"
                else -> word
            }
        }
}

/**
 * Проверяет, соответствует ли имя приложения поисковому запросу.
 */
fun matchesFuzzyQuery(appLabel: String, query: String): Boolean {
    val normalizedQuery = query.normalizeForSearch()
    val normalizedLabel = appLabel.normalizeForSearch()
    if (normalizedLabel.contains(normalizedQuery)) return true
    if (normalizedQuery.length <= 3) {
        val words = normalizedLabel.split(" ")
        if (words.any { it.startsWith(normalizedQuery) }) return true
    }
    return false
}

    color: Color = Color(0xFF1565C0),
    strokeWidth: Dp = 4.dp
) {
    val infiniteTransition = rememberInfiniteTransition(label = "morphing")
    val rotation by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = 360f,
        animationSpec = infiniteRepeatable(
            animation = tween(1200, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "rotation"
    )

    val morphAngle by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = 300f,
        animationSpec = infiniteRepeatable(
            animation = tween(800, easing = FastOutSlowInEasing),
            repeatMode = RepeatMode.Reverse
        ),
        label = "morphAngle"
    )

    Canvas(modifier = modifier.size(size)) {
        val stroke = Stroke(
            width = strokeWidth.toPx(),
            cap = StrokeCap.Round
        )
        drawArc(
            color = color,
            startAngle = rotation + morphAngle * 0.3f,
            sweepAngle = morphAngle,
            useCenter = false,
            style = stroke,
            size = Size(this.size.width, this.size.height)
        )
    }

        val currentPages = themeManager.homePageApps.value

        var newIndex = 0
        for (i in 0 until current) {
            if (i != pageIndex) {
                newMap[newIndex++] = currentPages[i] ?: emptyList()

        }
        themeManager.setHomePageAppsForAllPages(newMap, current - 1)

class SystemBarState {
    var isDashboard = mutableStateOf(false)
    var isWallpaperDark = mutableStateOf(true)
}

val LocalThemeSystemBars = staticCompositionLocalOf { SystemBarState() }
val LocalScale = compositionLocalOf { 1.0f }

@Composable
fun GluonTheme(
    themeMode: Int,
    dynamicColor: Boolean = true,
    scale: Float = 1.0f,
    systemBarState: SystemBarState = remember { SystemBarState() },
    content: @Composable () -> Unit
) {
    val context = LocalContext.current
    val view = LocalView.current

    val systemInDark = isSystemInDarkTheme()

    val darkTheme = remember(themeMode, systemInDark) {
        when (themeMode) {
            ThemeManager.MODE_LIGHT -> false
            ThemeManager.MODE_DARK -> true
            else -> systemInDark
        }
    }

    // ✅ Ключ dynamicColor теперь гарантирует перестроение схемы при переключении тумблера
    val colorScheme = remember(darkTheme, dynamicColor) {
        when {
            dynamicColor && Build.VERSION.SDK_INT >= Build.VERSION_CODES.S -> {
                if (darkTheme) dynamicDarkColorScheme(context) else dynamicLightColorScheme(context)
            }
            darkTheme -> GluonDarkColorScheme
            else -> GluonLightColorScheme
        }
    }

    val window = (view.context as Activity).window
    val insetsController = WindowCompat.getInsetsController(window, view)

    LaunchedEffect(darkTheme, systemBarState.isDashboard.value, systemBarState.isWallpaperDark.value) {
        WindowCompat.setDecorFitsSystemWindows(window, false)
        window.statusBarColor = android.graphics.Color.TRANSPARENT
        window.navigationBarColor = android.graphics.Color.TRANSPARENT

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            window.isStatusBarContrastEnforced = false
            window.isNavigationBarContrastEnforced = false
        }

        val useDarkIcons = if (systemBarState.isDashboard.value) {
            !systemBarState.isWallpaperDark.value
        } else {
            !darkTheme
        }

        insetsController.isAppearanceLightStatusBars = useDarkIcons
        insetsController.isAppearanceLightNavigationBars = useDarkIcons
    }

    CompositionLocalProvider(
        LocalScale provides scale,
        LocalThemeSystemBars provides systemBarState
    ) {
        MaterialTheme(
            colorScheme = colorScheme,
            typography = getGluonTypography(scale),
            shapes = GluonShapes,
            content = content
        )
    }
}

            expanded = expanded,
            onDismissRequest = { expanded = false },
            offset = touchPoint,
            containerColor = Color.Transparent,
            shadowElevation = 0.dp,
            modifier = Modifier.width((280 * scale).dp)

fun Context.toast(message: String, duration: Int = Toast.LENGTH_SHORT) {
    if (Looper.myLooper() == Looper.getMainLooper()) {
        Toast.makeText(this.applicationContext, message, duration).show()
    } else {
        Handler(Looper.getMainLooper()).post {
            Toast.makeText(this.applicationContext, message, duration).show()
        }
    }
}

fun Context.toast(@StringRes resId: Int, duration: Int = Toast.LENGTH_SHORT) {
    try {
        val message = getString(resId)
        toast(message, duration)
    } catch (e: Exception) {
        Log.e(TAG, "Resource not found", e)
    }
}

fun Context.openLink(url: String) {
    if (url.isBlank()) return
    try {
        val intent = Intent(Intent.ACTION_VIEW, url.trim().toUri()).apply {
            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        }
        startActivity(intent)
    } catch (e: Exception) {
        Log.e(TAG, "Failed to open link: $url", e)
        toast("Нет приложения для открытия ссылки")
    }
}

fun Context.openAppSettings(packageName: String) {
    try {
        val intent = Intent(android.provider.Settings.ACTION_APPLICATION_DETAILS_SETTINGS).apply {
            data = Uri.fromParts("package", packageName, null)
            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        }
        startActivity(intent)
    } catch (e: Exception) {
        Log.e(TAG, "Failed to open settings for $packageName", e)
        toast("Настройки недоступны")
    }
}

fun Context.launchApp(app: AppModel) {
    try {
        val intent = packageManager.getLaunchIntentForPackage(app.packageName)
        if (intent != null) {
            intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            startActivity(intent)
        } else {
            toast("Приложение ${app.label} не найдено")
        }
    } catch (e: Exception) {
        Log.e(TAG, "Launch error for ${app.packageName}", e)
        toast("Ошибка запуска")
    }
}

fun Intent.safeStart(context: Context) {
    try {
        if (this.flags and Intent.FLAG_ACTIVITY_NEW_TASK == 0) {
            this.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        }
        context.startActivity(this)
    } catch (e: Exception) {
        Log.e(TAG, "SafeStart failed", e)
        context.toast("Действие недоступно")
    }
}

fun String.formatGluonId(): String {
    val trimmed = this.trim().removePrefix("@")
    if (trimmed.isEmpty()) return ""
    return "@$trimmed"
}

fun String.isValidGluonId(): Boolean {
    val cleanId = this.removePrefix("@").trim()
    val regex = "^[a-zA-Z0-9_]{3,20}$".toRegex()
    return regex.matches(cleanId)
}

fun String.extractEmailPrefix(): String {
    val part = if (this.contains("@")) this.substringBefore("@") else this
    return part.filter { it.isLetterOrDigit() || it == '_' }
}

fun String.isValidEmail(): Boolean {
    val target = this.trim()
    val emailPattern = android.util.Patterns.EMAIL_ADDRESS
    return target.isNotBlank() &&
            emailPattern.matcher(target).matches() &&
            target.contains(".") &&
            !target.contains(",")
}

fun Context.copyToClipboard(label: String, text: String) {
    try {
        val clipboard = getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
        val clip = ClipData.newPlainText(label, text)
        clipboard.setPrimaryClip(clip)
        if (Build.VERSION.SDK_INT <= Build.VERSION_CODES.S_V2) {
            toast("Скопировано")
        }
    } catch (e: Exception) {
        Log.e(TAG, "Clipboard error", e)
        toast("Ошибка копирования")
    }
}

val Context.dataStore by preferencesDataStore(name = "gluon_app_repository")

private val json = Json { ignoreUnknownKeys = true }

private val USAGE_COUNT_KEY = stringPreferencesKey("usage_counts")
private val HIDDEN_APPS_KEY = stringPreferencesKey("hidden_apps")

suspend fun DataStore<Preferences>.getUsageCounts(): Map<String, Int> {
    return try {
        val stored = data.firstOrNull()?.get(USAGE_COUNT_KEY) ?: "{}"
        json.decodeFromString<Map<String, Int>>(stored)
    } catch (_: Exception) {
        emptyMap()
    }
}

suspend fun DataStore<Preferences>.incrementUsageCount(packageName: String) {
    edit { prefs ->
        val stored = prefs[USAGE_COUNT_KEY] ?: "{}"
        val current = try {
            json.decodeFromString<Map<String, Int>>(stored)
        } catch (_: Exception) {
            emptyMap()
        }
        val updated = current + (packageName to (current[packageName] ?: 0) + 1)
        prefs[USAGE_COUNT_KEY] = json.encodeToString(updated)
    }
}

suspend fun DataStore<Preferences>.getHiddenApps(): Set<String> {
    return try {
        val stored = data.firstOrNull()?.get(HIDDEN_APPS_KEY) ?: "[]"
        json.decodeFromString<List<String>>(stored).toSet()
    } catch (_: Exception) {
        emptySet()
    }
}

suspend fun DataStore<Preferences>.addHiddenApp(packageName: String) {
    edit { prefs ->
        val stored = prefs[HIDDEN_APPS_KEY] ?: "[]"
        val current = try {
            json.decodeFromString<List<String>>(stored)
        } catch (_: Exception) {
            emptyList()
        }
        val updated = current + packageName
        prefs[HIDDEN_APPS_KEY] = json.encodeToString(updated)
    }
}

suspend fun DataStore<Preferences>.removeHiddenApp(packageName: String) {
    edit { prefs ->
        val stored = prefs[HIDDEN_APPS_KEY] ?: "[]"
        val current = try {
            json.decodeFromString<List<String>>(stored)
        } catch (_: Exception) {
            emptyList()
        }
        val updated = current.filter { it != packageName }
        prefs[HIDDEN_APPS_KEY] = json.encodeToString(updated)
    }
}

import com.gluon.launcher.core.data.AppModel
import com.gluon.launcher.core.theme.LocalScale
import com.gluon.launcher.launcher.ui.components.GluonScaffold
import kotlinx.coroutines.delay

@Composable
fun SettingsScreen(
    authManager: AuthManager,
    onBack: () -> Unit,
    onLogout: () -> Unit,
    onEditProfile: () -> Unit,
    onDeleteAccount: () -> Unit,
    onNavigateToAuth: () -> Unit,
    onVerifyEmail: () -> Unit,
    onEditHomeScreen: () -> Unit, // новый колбэк для перехода к редактированию рабочего стола
    isProfileGlass: Boolean,
    showProfileAvatar: Boolean,
    onToggleProfileGlass: (Boolean) -> Unit,
    onToggleShowAvatar: (Boolean) -> Unit,
    isGlassFrosted: Boolean,
    onToggleGlassFrosted: (Boolean) -> Unit,
    themeMode: Int,
    isDynamicStyle: Boolean,
    onThemeModeChange: (Int) -> Unit,
    onDynamicStyleToggle: (Boolean) -> Unit,
    gridColumns: Int,
    onGridColumnsChange: (Int) -> Unit,
    isDockFrosted: Boolean,
    onDockFrostedToggle: (Boolean) -> Unit,
    isDrawerFrosted: Boolean,
    onDrawerFrostedToggle: (Boolean) -> Unit,
    showDockLabels: Boolean = false,
    onShowDockLabelsChange: (Boolean) -> Unit = {},
    showLabels: Boolean = true,
    onShowLabelsChange: (Boolean) -> Unit = {},
    showIconBorder: Boolean = false,
    onShowIconBorderChange: (Boolean) -> Unit = {},
    hiddenApps: Set<String> = emptySet(),
    hiddenAppModels: Map<String, AppModel> = emptyMap(),
    onUnhideApp: (String) -> Unit = {},
    appScale: Float = 1.0f,
    onAppScaleChange: (Float) -> Unit = {},
    isDockBarHidden: Boolean = false,
    onDockBarHiddenToggle: (Boolean) -> Unit = {}
) {
    val scale = LocalScale.current
    val user by authManager.currentUser.collectAsStateWithLifecycle()
    val isGuest by authManager.isGuest.collectAsStateWithLifecycle()
    val snackbarHostState = remember { SnackbarHostState() }
    var showDeleteDialog by remember { mutableStateOf(false) }
    var showLogoutDialog by remember { mutableStateOf(false) }
    var showDeveloperPopup by remember { mutableStateOf(false) }

    LaunchedEffect(showDeveloperPopup) {
        if (showDeveloperPopup) { delay(2000); showDeveloperPopup = false }
    }

    var profileExpanded by remember { mutableStateOf(false) }
    var appearanceExpanded by remember { mutableStateOf(false) }
    var homeScreenExpanded by remember { mutableStateOf(false) }
    var infoPanelExpanded by remember { mutableStateOf(false) }
    var dockExpanded by remember { mutableStateOf(false) }
    var drawerExpanded by remember { mutableStateOf(false) }
    var hiddenExpanded by remember { mutableStateOf(false) }

    val dynamicAvailable = Build.VERSION.SDK_INT >= Build.VERSION_CODES.S

    GluonScaffold(showBack = true, onBackClick = onBack) { modifier ->
        Box(modifier = modifier.fillMaxSize()) {
            val systemBarsPadding = WindowInsets.systemBars.asPaddingValues()
            LazyColumn(
                modifier = Modifier
                    .fillMaxSize()
                    .wrapContentWidth(Alignment.CenterHorizontally)
                    .widthIn(max = 600.dp),
                contentPadding = PaddingValues(
                    start = (30 * scale).dp,
                    end = (30 * scale).dp,
                    top = (60 * scale).dp,
                    bottom = systemBarsPadding.calculateBottomPadding()
                ),
                verticalArrangement = Arrangement.spacedBy((16 * scale).dp)
            ) {
                item {
                    Text(
                        "НАСТРОЙКИ",
                        style = MaterialTheme.typography.headlineMedium,
                        fontWeight = FontWeight.Black,
                        color = MaterialTheme.colorScheme.onBackground,
                        textAlign = TextAlign.Center,
                        modifier = Modifier.fillMaxWidth().padding(top = (30 * scale).dp, bottom = (24 * scale).dp)
                    )
                }

                // Профиль
                item {
                    ModernExpandableSection(
                        title = "Профиль",
                        icon = Icons.Default.AccountCircle,
                        expanded = profileExpanded,
                        onToggle = { profileExpanded = !profileExpanded },
                        alwaysVisible = {
                            if (isGuest) {
                                GuestPlaceholder(onNavigateToAuth)
                            } else {
                                ProfileHeader(
                                    name = user?.fullName ?: "Пользователь",
                                    gluonId = user?.gluonId ?: "—",
                                    avatarUrl = user?.getAvatarUrl()
                                )
                                if (user?.verified == false) VerificationBanner(onVerifyEmail)
                            }
                        }
                    ) {
                        if (!isGuest) {
                            SettingsActionRow("Редактировать профиль", Icons.Default.Edit, onClick = onEditProfile)
                            SettingsActionRow("Выйти из аккаунта", Icons.AutoMirrored.Filled.ExitToApp, color = MaterialTheme.colorScheme.primary, onClick = { showLogoutDialog = true })
                            SettingsActionRow("Удалить аккаунт", Icons.Default.DeleteForever, color = MaterialTheme.colorScheme.error, onClick = { showDeleteDialog = true })
                        }
                    }
                }

                // Оформление
                item {
                    ModernExpandableSection(
                        title = "Оформление",
                        icon = Icons.Default.Palette,
                        expanded = appearanceExpanded,
                        onToggle = { appearanceExpanded = !appearanceExpanded }
                    ) {
                        SettingsToggleRow(
                            label = "Динамическое оформление",
                            icon = Icons.Default.AutoAwesome,
                            checked = isDynamicStyle && dynamicAvailable,
                            enabled = dynamicAvailable,
                            onCheckedChange = onDynamicStyleToggle
                        )
                        ThemeModeSelector(currentMode = themeMode, onModeSelected = onThemeModeChange)
                        Spacer(Modifier.height((12 * scale).dp))

                        Row(
                            modifier = Modifier.fillMaxWidth(),
                            horizontalArrangement = Arrangement.SpaceBetween,
                            verticalAlignment = Alignment.CenterVertically
                        ) {
                            Row(verticalAlignment = Alignment.CenterVertically) {
                                Icon(Icons.Default.ZoomIn, null, Modifier.size((20 * scale).dp), tint = MaterialTheme.colorScheme.onSurfaceVariant)
                                Spacer(Modifier.width((12 * scale).dp))
                                Text("Масштаб интерфейса", style = MaterialTheme.typography.bodyLarge, fontWeight = FontWeight.Medium)
                            }
                            TextButton(onClick = { onAppScaleChange(0.90f) }) { Text("Сброс", fontWeight = FontWeight.Bold) }
                        }
                        Spacer(Modifier.height((4 * scale).dp))
                        Slider(value = appScale, onValueChange = onAppScaleChange, valueRange = 0.8f..1.0f, steps = 0, modifier = Modifier.fillMaxWidth())
                        Row(
                            modifier = Modifier.fillMaxWidth(),
                            horizontalArrangement = Arrangement.SpaceBetween,
                            verticalAlignment = Alignment.CenterVertically
                        ) {
                            Text("Минимальный", style = MaterialTheme.typography.bodySmall, color = MaterialTheme.colorScheme.onSurfaceVariant)
                            Text("По умолчанию", style = MaterialTheme.typography.bodySmall, color = MaterialTheme.colorScheme.onSurfaceVariant, modifier = Modifier.padding(start = (12 * scale).dp))
                            Text("Максимальный", style = MaterialTheme.typography.bodySmall, color = MaterialTheme.colorScheme.onSurfaceVariant)
                        }
                    }
                }

                item {
                    ModernExpandableSection(
                        title = "Рабочий стол",
                        icon = Icons.Default.Home,
                        expanded = homeScreenExpanded,
                        onToggle = { homeScreenExpanded = !homeScreenExpanded }
                    ) {
                        SettingsActionRow(
                            label = "Изменить рабочий стол",
                            icon = Icons.Default.Edit,
                            onClick = onEditHomeScreen
                        )
                        Spacer(Modifier.height((8 * scale).dp))
                        Text(
                            "Приложения можно добавлять на рабочий стол через режим редактирования (долгое нажатие на пустом месте).",
                            style = MaterialTheme.typography.bodySmall,
                            color = MaterialTheme.colorScheme.onSurfaceVariant
                        )
                    }

                        title = "Информационная панель",

                        expanded = infoPanelExpanded,

                        SettingsToggleRow("Прозрачный фон", Icons.Default.BlurOn, checked = isGlassFrosted, onCheckedChange = onToggleGlassFrosted)
                    }
                }

                // Док-бар
                item {
                    ModernExpandableSection(
                        title = "Док-бар",
                        icon = Icons.Default.List,
                        expanded = dockExpanded,
                        onToggle = { dockExpanded = !dockExpanded }
                    ) {

                // Меню приложений

                    ModernExpandableSection(
                        title = "Меню приложений",
                        icon = Icons.Default.Apps,
                        expanded = drawerExpanded,
                        onToggle = { drawerExpanded = !drawerExpanded }
                    ) {

                        SingleChoiceSegmentedButtonRow(Modifier.fillMaxWidth()) {

                        Spacer(Modifier.height((8 * scale).dp))

                            icon = Icons.Default.VisibilityOff,
                            expanded = hiddenExpanded,
                            onToggle = { hiddenExpanded = !hiddenExpanded }
                        ) {
                            hiddenApps.forEach { pkg ->
                                val model = hiddenAppModels[pkg]
                                Row(modifier = Modifier.fillMaxWidth().padding(vertical = (8 * scale).dp), verticalAlignment = Alignment.CenterVertically) {
                                    Box(modifier = Modifier.size((32 * scale).dp).clip(CircleShape).background(MaterialTheme.colorScheme.surfaceVariant), contentAlignment = Alignment.Center) {
                                        if (model?.iconBitmap != null) {
                                            Image(bitmap = model.iconBitmap, contentDescription = null, modifier = Modifier.fillMaxSize(), contentScale = ContentScale.Crop)
                                        } else {
                                            Text(text = model?.label?.take(1)?.uppercase() ?: "?", style = MaterialTheme.typography.bodySmall, fontWeight = FontWeight.Bold, color = MaterialTheme.colorScheme.onSurfaceVariant)
                                        }
                                    }
                                    Spacer(Modifier.width((12 * scale).dp))
                                    Text(text = model?.label ?: pkg, style = MaterialTheme.typography.bodyMedium, modifier = Modifier.weight(1f))
                                    Icon(imageVector = Icons.Default.Visibility, contentDescription = "Показать", modifier = Modifier.size((24 * scale).dp).clickable { onUnhideApp(pkg) }, tint = MaterialTheme.colorScheme.primary)
                                }
                            }
                        }
                    }
                }

                // О приложении
                item {
                    Column(
                        modifier = Modifier.padding(vertical = (8 * scale).dp, horizontal = (4 * scale).dp)
                            .combinedClickable(interactionSource = remember { MutableInteractionSource() }, indication = null, onClick = {}, onLongClick = { showDeveloperPopup = true })

                        Text("Gluon Launcher", style = MaterialTheme.typography.titleMedium, fontWeight = FontWeight.Bold, color = MaterialTheme.colorScheme.onSurface)
                        Spacer(Modifier.height((4 * scale).dp))
                        Text("Версия 7.1.0", style = MaterialTheme.typography.bodyMedium, color = MaterialTheme.colorScheme.onSurfaceVariant)
                    }
                }
            }

            SnackbarHost(hostState = snackbarHostState, modifier = Modifier.align(Alignment.BottomCenter))

            AnimatedVisibility(visible = showDeveloperPopup, enter = fadeIn(), exit = fadeOut(), modifier = Modifier.align(Alignment.BottomCenter).padding(bottom = (64 * scale).dp)) {
                Surface(modifier = Modifier.clip(RoundedCornerShape((16 * scale).dp)), color = MaterialTheme.colorScheme.primaryContainer, shadowElevation = 4.dp) {
                    Row(modifier = Modifier.padding(horizontal = (20 * scale).dp, vertical = (14 * scale).dp), verticalAlignment = Alignment.CenterVertically) {
                        Icon(Icons.Default.Info, null, tint = MaterialTheme.colorScheme.primary, modifier = Modifier.size((28 * scale).dp))
                        Spacer(Modifier.width((12 * scale).dp))
                        Text("Разработчик REIMS", style = MaterialTheme.typography.titleMedium, fontWeight = FontWeight.Bold, color = MaterialTheme.colorScheme.onPrimaryContainer)
                    }
                }
            }

            if (showLogoutDialog) {
                AlertDialog(
                    onDismissRequest = { showLogoutDialog = false },
                    icon = { Icon(Icons.AutoMirrored.Filled.ExitToApp, null, tint = MaterialTheme.colorScheme.primary) },
                    title = { Text("Выход из аккаунта", fontWeight = FontWeight.Bold) },
                    text = { Text("Вы уверены, что хотите выйти?") },
                    confirmButton = { Button(onClick = { showLogoutDialog = false; onLogout() }, shape = RoundedCornerShape((12 * scale).dp)) { Text("ВЫЙТИ", fontWeight = FontWeight.Bold) } },
                    dismissButton = { TextButton(onClick = { showLogoutDialog = false }) { Text("ОТМЕНА") } },
                    shape = MaterialTheme.shapes.extraLarge
                )
            }
            if (showDeleteDialog) {
                AlertDialog(
                    onDismissRequest = { showDeleteDialog = false },
                    icon = { Icon(Icons.Default.Warning, null, tint = MaterialTheme.colorScheme.error) },
                    title = { Text("Удаление аккаунта", fontWeight = FontWeight.Bold) },
                    text = { Text("Это действие необратимо. Все данные GluonCore будут удалены.") },
                    confirmButton = { Button(onClick = { showDeleteDialog = false; onDeleteAccount() }, colors = ButtonDefaults.buttonColors(containerColor = MaterialTheme.colorScheme.error), shape = RoundedCornerShape((12 * scale).dp)) { Text("УДАЛИТЬ", fontWeight = FontWeight.Bold) } },
                    dismissButton = { TextButton(onClick = { showDeleteDialog = false }) { Text("ОТМЕНА") } },
                    shape = MaterialTheme.shapes.extraLarge
                )
            }
        }
    }
}

@Composable
fun EditModeHomeGrid(
    appsList: List<String?>,
    allApps: List<AppModel>,
    totalRows: Int,
    columns: Int,
    iconScale: Float = 1f,
    onAddAppClicked: (Int) -> Unit,
    onRemoveApp: (Int) -> Unit,
    textColor: Color = Color.White   // ✅ новый параметр, по умолчанию белый для тёмного фона редактирования
) {
    val scale = LocalScale.current
    val horizontalPadding = (16 * scale).dp
    val horizontalSpacing = (when (columns) { 3 -> 16; 4 -> 12; 5 -> 8; else -> 12 } * scale).dp
    val verticalSpacing = (18 * scale).dp

    Column(
        modifier = Modifier.fillMaxSize(),
        verticalArrangement = Arrangement.spacedBy(verticalSpacing)
    ) {
        for (row in 0 until totalRows) {
            Row(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxWidth()
                    .padding(horizontal = horizontalPadding),
                horizontalArrangement = Arrangement.spacedBy(horizontalSpacing)
            ) {
                for (col in 0 until columns) {
                    val index = row * columns + col
                    val pkg = if (index < appsList.size) appsList[index] else null
                    val app = if (pkg != null) allApps.find { it.packageName == pkg } else null
                    val isOccupied = app != null
                    Box(
                        modifier = Modifier
                            .weight(1f)
                            .fillMaxSize()
                            .padding((8 * scale).dp)
                            .clip(CircleShape)
                            .then(
                                if (!isOccupied) {
                                    Modifier
                                        .border(1.dp, Color.White.copy(alpha = 0.5f), CircleShape)
                                        .clickable { onAddAppClicked(index) }
                                } else Modifier
                            ),
                        contentAlignment = Alignment.Center
                    ) {
                        if (isOccupied) {
                            AppIconItem(
                                app = app!!,
                                scale = iconScale,
                                showLabel = false,          // в режиме редактирования подписи скрыты
                                showIconBorder = false,
                                isInGrid = true,
                                onAppClick = { /* no click */ },
                                modifier = Modifier.fillMaxSize(),
                                textColor = textColor,     // ✅ передаём цвет (будет белым)
                                onContextMenu = { dismiss ->
                                    Column(
                                        modifier = Modifier.width((180 * scale).dp),
                                        verticalArrangement = Arrangement.spacedBy(4.dp)
                                    ) {
                                        GluonPopupMenuCard(
                                            onClick = { dismiss(); onRemoveApp(index) },
                                            color = MaterialTheme.colorScheme.errorContainer
                                        ) {
                                            Icon(Icons.Default.Delete, null, tint = MaterialTheme.colorScheme.onErrorContainer)
                                            Spacer(Modifier.width(8.dp))
                                            GluonMarqueeText("Убрать с рабочего стола", color = MaterialTheme.colorScheme.onErrorContainer)
                                        }
                                    }
                                }
                            )
                        } else {
                            Icon(
                                Icons.Default.Add,
                                contentDescription = "Добавить",
                                tint = Color.White.copy(alpha = 0.7f),
                                modifier = Modifier.size((32 * scale).dp)
                            )
                        }
                    }
                }
            }
        }
    }
}

@Composable
fun AuthTextField(
    value: String,
    onValueChange: (String) -> Unit,
    label: String,
    icon: ImageVector,
    enabled: Boolean = true,
    isError: Boolean = false,
    supportingText: String? = null,
    keyboardType: KeyboardType = KeyboardType.Text,
    isPassword: Boolean = false,
    passwordVisible: Boolean = false,
    onPasswordToggle: () -> Unit = {},
    imeAction: ImeAction = ImeAction.Next,
    onDone: (() -> Unit)? = null
) {
    val scale = LocalScale.current
    val focusManager = LocalFocusManager.current

    TextField(
        value = value,
        onValueChange = onValueChange,
        label = { Text(label, fontSize = (14 * scale).sp) },
        leadingIcon = { Icon(icon, null, Modifier.size((22 * scale).dp)) },
        modifier = Modifier

@Composable
fun MainDockBar(
    dockPackages: List<String>,
    allApps: List<AppModel>,
    showIconBorder: Boolean,
    showDockLabels: Boolean = false,
    onAppClick: (AppModel) -> Unit,
    onRemoveFromDock: (String) -> Unit,
    onMoveLeft: (String) -> Unit,
    onMoveRight: (String) -> Unit,
    onHideApp: (String) -> Unit,
    onSwipeUp: () -> Unit,
    isDockFrosted: Boolean = false,
    onToggleDockFrosted: (Boolean) -> Unit = {},
    isDockBarHidden: Boolean = false,
    onToggleDockBarHidden: (Boolean) -> Unit = {},
    modifier: Modifier = Modifier,
    frost: Boolean = false,
    maxDockApps: Int = 5,
    isEditMode: Boolean = false,
    customHeight: Dp = 85.dp,
    isEditModePanel: Boolean = false
) {
    val isVisible = !isDockBarHidden && !isEditMode
    val showEditFrame = isEditMode && !isDockBarHidden && isEditModePanel
    if (!isVisible && !showEditFrame) return

    val scale = LocalScale.current
    val context = LocalContext.current

    val slots = dockPackages.take(maxDockApps).mapNotNull { pkg -> allApps.find { it.packageName == pkg } }
    val shape = RoundedCornerShape(if (isEditMode && isEditModePanel) 50.dp else M3EShapes.ExtraExtraLarge)

    var showDockMenu by remember { mutableStateOf(false) }
    var dockMenuOffset by remember { mutableStateOf(DpOffset.Zero) }

    val dockIconScale = if (maxDockApps >= 5) scale * 0.85f else scale

    val horizontalPadding = (16 * scale).dp
    val verticalPadding = (16 * scale).dp
    val spacing = (when (maxDockApps) { 3 -> 16; 4 -> 12; 5 -> 8; else -> 12 } * scale).dp

    // Блюр для подложки (frost или isDockFrosted)
    val showGlass = (frost || isDockFrosted) && !isEditMode && Build.VERSION.SDK_INT >= Build.VERSION_CODES.S
    val blurEffect = if (showGlass) {
        RenderEffect.createBlurEffect(25f, 25f, Shader.TileMode.DECAL).asComposeRenderEffect()
    } else null

    Box(
        modifier = modifier
            .wrapContentHeight()
            .then(if (!isEditMode) Modifier.padding(bottom = (8 * scale).dp) else Modifier)
            .pointerInput(Unit) { if (!isEditMode) detectVerticalDragGestures { _, dragAmount -> if (dragAmount < -50f * scale) onSwipeUp() } }
            .pointerInput(Unit) { if (!isEditMode) detectTapGestures(onLongPress = { offset -> dockMenuOffset = DpOffset(offset.x.toDp(), offset.y.toDp()); showDockMenu = true }) },
        contentAlignment = Alignment.BottomCenter
    ) {
        val contentColor = if (frost || isDockFrosted) Color.White else MaterialTheme.colorScheme.onSurface
        CompositionLocalProvider(LocalContentColor provides contentColor) {
            if (showEditFrame) {
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(vertical = (10 * scale).dp)
                        .wrapContentHeight()
                        .background(Color.Transparent, shape)
                        .border(width = 2.dp, color = Color.White, shape = shape)
                        .clip(shape),
                    contentAlignment = Alignment.Center
                ) {
                    Box(modifier = Modifier.fillMaxWidth().height(customHeight))
                }
            } else if (isVisible) {
                // Основной контейнер док-бара
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .wrapContentHeight()
                        .clip(shape)
                ) {
                    // 1. Размытая подложка (если frost)
                    if (showGlass) {
                        Surface(
                            modifier = Modifier
                                .matchParentSize()
                                .graphicsLayer(renderEffect = blurEffect),
                            shape = shape,
                            color = Color.White.copy(alpha = 0.15f)
                        ) {}
                    }

                    // 2. Контент (без блюра)
                    Box(
                        modifier = Modifier
                            .fillMaxWidth()
                            .wrapContentHeight()
                            .shadow(elevation = 0.dp, shape = shape, clip = false)
                            .border(
                                BorderStroke(
                                    0.5.dp,
                                    if (frost || isDockFrosted) Color.White.copy(alpha = 0.2f) else MaterialTheme.colorScheme.outlineVariant.copy(alpha = 0.3f)
                                ), shape
                            )
                            .background(
                                if (frost || isDockFrosted) Color.Transparent else MaterialTheme.colorScheme.surface,
                                shape
                            )
                            .clip(shape),
                        contentAlignment = Alignment.Center
                    ) {
                        if (slots.isNotEmpty()) {
                            Row(
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .padding(horizontal = horizontalPadding, vertical = verticalPadding),
                                horizontalArrangement = Arrangement.spacedBy(spacing, Alignment.CenterHorizontally),
                                verticalAlignment = Alignment.CenterVertically
                            ) {
                                slots.forEach { app ->
                                    Box(
                                        modifier = Modifier
                                            .weight(1f)
                                            .wrapContentSize(Alignment.Center),
                                        contentAlignment = Alignment.Center
                                    ) {
                                        AppIconItem(
                                            app = app,
                                            scale = dockIconScale,
                                            showLabel = showDockLabels,
                                            showIconBorder = showIconBorder,
                                            isInDockBar = true,
                                            onAppClick = onAppClick,
                                            modifier = Modifier,
                                            enableShadow = true,
                                            contextMenuOffset = DpOffset((16 * scale).dp, 0.dp),
                                            onContextMenu = { dismiss ->
                                                Column(
                                                    modifier = Modifier.width((180 * scale).dp),
                                                    verticalArrangement = Arrangement.spacedBy(4.dp)
                                                ) {
                                                    Row(
                                                        modifier = Modifier.fillMaxWidth(),
                                                        horizontalArrangement = Arrangement.spacedBy(4.dp)
                                                    ) {
                                                        GluonPopupMenuCard(onClick = { dismiss(); onMoveLeft(app.packageName) }, modifier = Modifier.weight(1f), color = MaterialTheme.colorScheme.surfaceVariant) {
                                                            Text("<", modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center, fontSize = (20 * scale).sp, fontWeight = FontWeight.Bold, color = MaterialTheme.colorScheme.onSurface)
                                                        }
                                                        GluonPopupMenuCard(onClick = { dismiss(); onRemoveFromDock(app.packageName) }, modifier = Modifier.weight(1f), color = MaterialTheme.colorScheme.errorContainer) {
                                                            Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) { Icon(Icons.Default.Close, null, tint = MaterialTheme.colorScheme.onErrorContainer) }
                                                        }
                                                        GluonPopupMenuCard(onClick = { dismiss(); onMoveRight(app.packageName) }, modifier = Modifier.weight(1f), color = MaterialTheme.colorScheme.surfaceVariant) {
                                                            Text(">", modifier = Modifier.fillMaxWidth(), textAlign = TextAlign.Center, fontSize = (20 * scale).sp, fontWeight = FontWeight.Bold, color = MaterialTheme.colorScheme.onSurface)
                                                        }
                                                    }
                                                    GluonPopupMenuCard(
                                                        onClick = {
                                                            dismiss()
                                                            context.startActivity(Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS).apply { data = Uri.fromParts("package", app.packageName, null) })
                                                        },
                                                        color = MaterialTheme.colorScheme.surfaceContainerHigh
                                                    ) {
                                                        Icon(Icons.Default.Info, null, tint = MaterialTheme.colorScheme.onSurface)
                                                        Spacer(Modifier.width(8.dp))
                                                        GluonMarqueeText("О приложении")
                                                    }
                                                }
                                            }
                                        )
                                    }
                                }
                            }
                        } else {
                            Box(
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .height(customHeight)
                                    .padding(horizontal = horizontalPadding, vertical = verticalPadding),
                                contentAlignment = Alignment.Center
                            ) {
                                Text(
                                    text = "Нет приложений в док-баре. Добавьте через меню приложений",
                                    style = MaterialTheme.typography.bodySmall,
                                    color = contentColor.copy(alpha = 0.6f),
                                    textAlign = TextAlign.Center
                                )
                            }
                        }
                    }
                }
            }
        }

        if (!isEditMode && !isDockBarHidden) {
            DropdownMenu(
                expanded = showDockMenu,
                onDismissRequest = { showDockMenu = false },
                offset = dockMenuOffset,
                containerColor = Color.Transparent,
                shadowElevation = 0.dp,
                modifier = Modifier.width((250 * scale).dp)
            ) {
                GluonPopupMenuCard(onClick = { /* handled by Switch */ }, color = MaterialTheme.colorScheme.surfaceContainerHigh) {
                    Icon(Icons.Default.BlurOn, null, Modifier.size((20 * scale).dp))
                    Spacer(Modifier.width(12.dp))
                    GluonMarqueeText("Прозрачный фон", modifier = Modifier.weight(1f))
                    Switch(checked = isDockFrosted, onCheckedChange = { onToggleDockFrosted(it); showDockMenu = false }, modifier = Modifier.scale(0.7f))
                }
                Spacer(Modifier.height(4.dp))
                GluonPopupMenuCard(onClick = { /* handled by Switch */ }, color = MaterialTheme.colorScheme.surfaceContainerHigh) {
                    Icon(Icons.Default.Home, null, Modifier.size((20 * scale).dp))
                    Spacer(Modifier.width(12.dp))
                    GluonMarqueeText("Скрыть док-бар", modifier = Modifier.weight(1f))
                    Switch(checked = isDockBarHidden, onCheckedChange = { onToggleDockBarHidden(it); showDockMenu = false }, modifier = Modifier.scale(0.7f))
                }
            }
        }
    }

class MainActivity : ComponentActivity() {

    private val viewModel: MainViewModel by viewModels { MainViewModelFactory(application) }

    override fun onCreate(savedInstanceState: Bundle?) {
        enableEdgeToEdge()
        super.onCreate(savedInstanceState)

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            window.apply {
                addFlags(WindowManager.LayoutParams.FLAG_BLUR_BEHIND)
                setBackgroundBlurRadius((60 * resources.displayMetrics.density).toInt())
            }
        }

        @Suppress("DEPRECATION")
        window.setSoftInputMode(WindowManager.LayoutParams.SOFT_INPUT_ADJUST_RESIZE)

        setContent {
            val isReady by viewModel.isReady.collectAsStateWithLifecycle()
            val themeMode by viewModel.themeMode.collectAsStateWithLifecycle()
            val dynamicEnabled by viewModel.dynamicColors.collectAsStateWithLifecycle()
            val appScale by viewModel.appScale.collectAsStateWithLifecycle()
            val appState by viewModel.appState.collectAsStateWithLifecycle()
            val allApps by viewModel.allApps.collectAsStateWithLifecycle()
            val settingsState by viewModel.settingsState.collectAsStateWithLifecycle()

            val systemBarState = remember { SystemBarState() }

            GluonTheme(
                themeMode = themeMode,
                dynamicColor = dynamicEnabled,
                scale = appScale,
                systemBarState = systemBarState
            ) {
                if (!isReady) {
                    Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                        GluonMorphingLoader()
                    }
                } else {
                    AppNavigation(
                        appState = appState,
                        viewModel = viewModel,
                        authManager = viewModel.authManager,
                        profileManager = viewModel.profileManager,
                        settingsState = settingsState,
                        allApps = allApps,
                        onStateChange = viewModel::setAppState
                    )
                }
            }
        }
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        setIntent(intent)
        if (Intent.ACTION_MAIN == intent.action && intent.hasCategory(Intent.CATEGORY_HOME)) {
            viewModel.onHomePressed()
        }
    }

    override fun onResume() {
        super.onResume()
        lifecycleScope.launch {
            viewModel.appRepository.refreshApps()
        }
    }

    override fun onDestroy() {
        viewModel.appRepository.onDestroy()
        super.onDestroy()
    }
}

package com.gluon.launcher.theme

import android.content.Context
import android.content.Intent
import android.content.SharedPreferences
import androidx.core.content.edit
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.serialization.encodeToString
import kotlinx.serialization.json.Json

class ThemeManager(context: Context) {

    private val appContext = context.applicationContext
    private val prefs: SharedPreferences = appContext.getSharedPreferences("gluon_settings", Context.MODE_PRIVATE)
    private val json = Json { ignoreUnknownKeys = true }

    // ========== StateFlow для всех настроек ==========
    private val _appScale = MutableStateFlow(prefs.getFloat("app_scale", autoScale()).coerceIn(0.8f, 1.0f))
    val appScale: StateFlow<Float> = _appScale.asStateFlow()

    private val _themeMode = MutableStateFlow(prefs.getInt("theme_mode", MODE_SYSTEM))
    val themeMode: StateFlow<Int> = _themeMode.asStateFlow()

    private val _dynamicColors = MutableStateFlow(prefs.getBoolean("dynamic_color", true))
    val dynamicColors: StateFlow<Boolean> = _dynamicColors.asStateFlow()

    private val _isProfileGlass = MutableStateFlow(prefs.getBoolean("is_profile_glass", false))
    val isProfileGlass: StateFlow<Boolean> = _isProfileGlass.asStateFlow()

    private val _showProfileAvatar = MutableStateFlow(prefs.getBoolean("show_profile_avatar", true))
    val showProfileAvatar: StateFlow<Boolean> = _showProfileAvatar.asStateFlow()

    private val _isGlassFrosted = MutableStateFlow(prefs.getBoolean("is_glass_frosted", false))
    val isGlassFrosted: StateFlow<Boolean> = _isGlassFrosted.asStateFlow()

    private val _isDockFrosted = MutableStateFlow(prefs.getBoolean("dock_frosted", false))
    val isDockFrosted: StateFlow<Boolean> = _isDockFrosted.asStateFlow()

    private val _isDrawerFrosted = MutableStateFlow(prefs.getBoolean("drawer_frosted", false))
    val isDrawerFrosted: StateFlow<Boolean> = _isDrawerFrosted.asStateFlow()

    private val _gridColumns = MutableStateFlow(prefs.getInt("grid_columns", 4).coerceIn(3, 5))
    val gridColumns: StateFlow<Int> = _gridColumns.asStateFlow()

    private val _showLabels = MutableStateFlow(prefs.getBoolean("show_labels", true))
    val showLabels: StateFlow<Boolean> = _showLabels.asStateFlow()

    private val _showDockLabels = MutableStateFlow(prefs.getBoolean("dock_labels", false))
    val showDockLabels: StateFlow<Boolean> = _showDockLabels.asStateFlow()

    private val _showIconBorder = MutableStateFlow(prefs.getBoolean("icon_border", false))
    val showIconBorder: StateFlow<Boolean> = _showIconBorder.asStateFlow()

    private val _isDockBarHidden = MutableStateFlow(prefs.getBoolean("dock_bar_hidden", false))
    val isDockBarHidden: StateFlow<Boolean> = _isDockBarHidden.asStateFlow()

    private val _dockApps = MutableStateFlow<List<String>>(emptyList())
    val dockApps: StateFlow<List<String>> = _dockApps.asStateFlow()

    // ========== инициализация ==========
    init {
        loadDockApps()
        if (!prefs.contains("is_profile_glass")) {
            prefs.edit { putBoolean("is_profile_glass", false) }
        }
        if (!prefs.contains("dock_bar_hidden")) {
            prefs.edit { putBoolean("dock_bar_hidden", false) }
        }
        if (_dockApps.value.isEmpty()) {
            val defaultApps = getDefaultDockApps()
            if (defaultApps.isNotEmpty()) {
                setDockApps(defaultApps)
            }
        }
    }

    private fun loadDockApps() {
        val saved = prefs.getString("dock_apps", null)
        val list = if (!saved.isNullOrBlank()) saved.split(",").filter { it.isNotBlank() } else emptyList()
        _dockApps.value = list
    }

    private fun autoScale(): Float {
        val metrics = appContext.resources.displayMetrics
        val density = metrics.density
        val widthDp = metrics.widthPixels / density
        val heightDp = metrics.heightPixels / density
        val minSideDp = minOf(widthDp, heightDp)
        return when {
            density <= 1.5f || minSideDp <= 360f -> 0.85f
            density >= 3.5f -> 1.0f
            else -> 0.90f
        }
    }

    // ========== настройки профиля ==========
    fun setIsProfileGlass(glass: Boolean) {
        if (_isProfileGlass.value == glass) return
        prefs.edit { putBoolean("is_profile_glass", glass) }
        _isProfileGlass.value = glass
    }

    fun setShowProfileAvatar(show: Boolean) {
        if (_showProfileAvatar.value == show) return
        prefs.edit { putBoolean("show_profile_avatar", show) }
        _showProfileAvatar.value = show
    }

    fun setIsGlassFrosted(frosted: Boolean) {
        if (_isGlassFrosted.value == frosted) return
        prefs.edit { putBoolean("is_glass_frosted", frosted) }
        _isGlassFrosted.value = frosted
    }

    // ========== док-бар ==========
    private fun getDefaultDockApps(): List<String> {
        val pm = appContext.packageManager
        val result = mutableListOf<String>()
        val intents = listOf(
            Intent(Intent.ACTION_DIAL).apply { addCategory(Intent.CATEGORY_DEFAULT) },
            Intent(Intent.ACTION_VIEW, android.provider.ContactsContract.Contacts.CONTENT_URI).apply { addCategory(Intent.CATEGORY_DEFAULT) },
            Intent(android.provider.MediaStore.ACTION_IMAGE_CAPTURE),
            Intent(Intent.ACTION_VIEW, android.net.Uri.parse("http://")).apply { addCategory(Intent.CATEGORY_BROWSABLE) }
        )
        for (intent in intents) {
            if (result.size >= 4) break
            val pkg = pm.queryIntentActivities(intent, 0).firstOrNull()?.activityInfo?.packageName
            if (pkg != null && pkg != appContext.packageName) result.add(pkg)
        }
        return result
    }

    fun setDockApps(packages: List<String>) {
        val newList = packages.take(5)
        _dockApps.value = newList
        prefs.edit { putString("dock_apps", newList.joinToString(",")) }
    }

    fun setIsDockFrosted(frosted: Boolean) {
        if (_isDockFrosted.value == frosted) return
        prefs.edit { putBoolean("dock_frosted", frosted) }
        _isDockFrosted.value = frosted
    }

    fun setIsDockBarHidden(hidden: Boolean) {
        if (_isDockBarHidden.value == hidden) return
        prefs.edit { putBoolean("dock_bar_hidden", hidden) }
        _isDockBarHidden.value = hidden
    }

    // ========== меню приложений ==========
    fun setGridColumns(columns: Int) {
        val clamped = columns.coerceIn(3, 5)
        if (_gridColumns.value == clamped) return
        prefs.edit { putInt("grid_columns", clamped) }
        _gridColumns.value = clamped
    }

    fun setShowLabels(show: Boolean) {
        if (_showLabels.value == show) return
        prefs.edit { putBoolean("show_labels", show) }
        _showLabels.value = show
    }

    fun setShowDockLabels(show: Boolean) {
        if (_showDockLabels.value == show) return
        prefs.edit { putBoolean("dock_labels", show) }

    val homePagesCount: StateFlow<Int> = _homePagesCount.asStateFlow()

    private val _homePageApps = MutableStateFlow<Map<Int, List<String?>>>(emptyMap())
    val homePageApps: StateFlow<Map<Int, List<String?>>> = _homePageApps.asStateFlow()

    init {
        refreshHomePagesData()
    }

    private fun refreshHomePagesData() {
        val count = _homePagesCount.value
        val map = mutableMapOf<Int, List<String?>>()
        for (i in 0 until count) {
            val key = "home_page_$i"
            val saved = prefs.getString(key, null)
            val list = if (!saved.isNullOrBlank()) {
                try { json.decodeFromString<List<String?>>(saved) } catch (_: Exception) { emptyList() }
            } else {
                emptyList()
            }
            map[i] = list
        }
        _homePageApps.value = map
    }

    fun setHomePagesCount(count: Int) {
        val clamped = count.coerceIn(1, 5)
        if (_homePagesCount.value == clamped) return
        prefs.edit { putInt("home_pages_count", clamped) }
        _homePagesCount.value = clamped
        refreshHomePagesData()
    }

    fun getHomePageApps(pageIndex: Int): List<String?> = _homePageApps.value[pageIndex] ?: emptyList()

    private fun setHomePageApps(pageIndex: Int, apps: List<String?>) {
        val key = "home_page_$pageIndex"
        prefs.edit { putString(key, json.encodeToString(apps)) }
    }

    fun addAppToHomePage(pageIndex: Int, packageName: String, position: Int) {
        val current = getHomePageApps(pageIndex).toMutableList()
        if (position < 0 || position > current.size) return
        if (position == current.size) {
            current.add(packageName)
        } else {
            if (current[position] == null) current[position] = packageName
            else return
        }
        setHomePageApps(pageIndex, current)
        refreshHomePagesData()
    }

    fun removeAppFromHomePage(pageIndex: Int, position: Int) {
        val current = getHomePageApps(pageIndex).toMutableList()
        if (position in current.indices && current[position] != null) {
            current[position] = null
            setHomePageApps(pageIndex, current)
            refreshHomePagesData()
        }
    }

    fun resetHomePages() {
        val count = _homePagesCount.value
        for (i in 0 until count) {
            setHomePageApps(i, emptyList())
        }
        refreshHomePagesData()
    }

    fun setHomePageAppsForAllPages(pagesMap: Map<Int, List<String?>>, newCount: Int) {
        for ((index, apps) in pagesMap) {
            val key = "home_page_$index"
            prefs.edit { putString(key, json.encodeToString(apps)) }
        }
        val clampedCount = newCount.coerceIn(1, 5)
        prefs.edit { putInt("home_pages_count", clampedCount) }
        _homePagesCount.value = clampedCount

        val map = mutableMapOf<Int, List<String?>>()
        for (i in 0 until clampedCount) {
            val key = "home_page_$i"
            val saved = prefs.getString(key, null)
            val list = if (!saved.isNullOrBlank()) {
                try { json.decodeFromString<List<String?>>(saved) } catch (_: Exception) { emptyList() }
            } else {
                emptyList()
            }
            map[i] = list
        }
        _homePageApps.value = map
    }

        val list = _dockApps.value.toMutableList()
        val index = list.indexOf(packageName)
        if (index > 0) {
            list[index] = list[index - 1]
            list[index - 1] = packageName
            setDockApps(list)
        }
    }

    fun moveDockAppRight(packageName: String) {
        val list = _dockApps.value.toMutableList()
        val index = list.indexOf(packageName)
        if (index != -1 && index < list.size - 1) {
            list[index] = list[index + 1]
            list[index + 1] = packageName
            setDockApps(list)
        }
    }
